cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(text-to-svg VERSION 0.1.2 LANGUAGES C CXX)

include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
foreach(feature CAIRO FREETYPE GRAPHITE2 GLIB ICU CORETEXT UNISCRIBE GDI DIRECTWRITE GOBJECT INTROSPECTION)
  set(HB_HAVE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(HB_BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(HB_BUILD_SUBSET OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
FetchContent_Declare(freetype
  URL https://codeload.github.com/freetype/freetype/tar.gz/42608f77f20749dd6ddc9e0536788eaad70ea4b5
  URL_HASH SHA256=68ce87bb59ea209eb7350f41a94a27519ce16b37011b475a1e62d4abee154b66)
FetchContent_Declare(harfbuzz
  URL https://codeload.github.com/harfbuzz/harfbuzz/tar.gz/7b27c8edd46c674e01dd226fa9e1aa7549f5c436
  URL_HASH SHA256=818ac15af42769d5806d5f53077664976790264a8fd9161e135904b92ca9a923)
FetchContent_Declare(nlohmann_json
  URL https://codeload.github.com/nlohmann/json/tar.gz/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03
  URL_HASH SHA256=0dbc5e40a01ff142e7e68c03e85247a4dcede2f592d12d3677dee3664d17975a)
FetchContent_MakeAvailable(freetype harfbuzz nlohmann_json)

add_library(outline-engine STATIC outline.cpp)
target_compile_features(outline-engine PUBLIC cxx_std_17)
target_include_directories(outline-engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(outline-engine PUBLIC nlohmann_json::nlohmann_json PRIVATE freetype harfbuzz)
add_executable(text-to-svg main.cpp)
target_compile_definitions(text-to-svg PRIVATE TEXTSVG_VERSION="${PROJECT_VERSION}")
target_link_libraries(text-to-svg PRIVATE outline-engine)
set_target_properties(outline-engine text-to-svg PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(outline-engine PRIVATE /utf-8 /W4)
  target_compile_options(text-to-svg PRIVATE /utf-8)
elseif(MINGW)
  target_link_options(text-to-svg PRIVATE -municode)
  target_link_options(text-to-svg PRIVATE -static -static-libgcc -static-libstdc++)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(outline-engine-tests engine_tests.cpp)
  target_link_libraries(outline-engine-tests PRIVATE outline-engine)
  if(MSVC)
    target_compile_options(outline-engine-tests PRIVATE /utf-8)
  endif()
  add_test(NAME outline-engine COMMAND outline-engine-tests)
endif()
